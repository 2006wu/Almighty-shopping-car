cmake_minimum_required(VERSION 3.8)
project(shopping_microros_client C)
find_package(ament_cmake REQUIRED)
find_package(rcl REQUIRED)
find_package(rclc REQUIRED)
find_package(rmw_microxrcedds REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(rosidl_runtime_c REQUIRED)
add_executable(sim_client main.c model.c)
target_compile_features(sim_client PRIVATE c_std_99)
target_compile_options(sim_client PRIVATE -Wall -Wextra)
ament_target_dependencies(sim_client rcl rclc rmw_microxrcedds geometry_msgs sensor_msgs std_msgs rosidl_runtime_c)
target_link_libraries(sim_client m)
install(TARGETS sim_client DESTINATION lib/${PROJECT_NAME})
ament_package()
