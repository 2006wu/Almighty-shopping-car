cmake_minimum_required(VERSION 3.8)
project(shopping_robot_gazebo)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
ament_python_install_package(shopping_robot_sim)
install(PROGRAMS shopping_robot_sim/fault_injector.py DESTINATION lib/${PROJECT_NAME} RENAME fault_injector)
install(PROGRAMS shopping_robot_sim/pedestrian_mover.py DESTINATION lib/${PROJECT_NAME} RENAME pedestrian_mover)
install(PROGRAMS shopping_robot_sim/initial_pose_node.py DESTINATION lib/${PROJECT_NAME} RENAME scenario_initial_pose)
foreach(dir launch config urdf worlds rviz maps)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/${dir}")
    install(DIRECTORY ${dir} DESTINATION share/${PROJECT_NAME})
  endif()
endforeach()
ament_package()
